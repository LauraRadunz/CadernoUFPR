      alert(n+" itens de progresso importados. A página vai recarregar.");location.reload();}catch(e){alert("Arquivo inválido.");}};r.readAsText(f);};
  inp.click();}
function lastInFlow(){return st.mode==="rev"?st.qp+1>=st.queue.length:st.i+1>=QS.length;}
function question(){
  var q=QS[st.i],a=aulaOf(q.aula),pos=posIn(q),cnt=stats(a.n).t;
  var h=top()+progress(st.i)+'<section class="folio"><div class="q-head"><div class="eyebrow">'+(st.mode==="rev"?'Revisão · ':'')+'Questão '+pos+' de '+cnt+' · '+cad.un+' '+a.rom+'</div>'+(q.tag?'<span class="tag">'+esc(q.tag)+'</span>':'')+'</div>'+stem(q)+'<ul class="opts'+(q.imgq?' letters':'')+'">';
  if(q.type==="soma"){
    q.opts.forEach(function(o,k){var v=q.vals[k],on=((st.sel||0)&v)!==0,inAns=(q.ans&v)!==0,c="opt";
      if(st.done){if(inAns)c+=" right";else if(on)c+=" wrong";}else if(on)c+=" sel";
      var m=st.done?(inAns?'<span class="mark">Soma da resposta</span>':(on?'<span class="mark">Marcada por você</span>':'')):'';
      h+='<li><button class="'+c+'" data-a="bit" data-v="'+v+'"'+(st.done?' disabled':'')+' aria-pressed="'+on+'"><span class="l">'+(v<10?'0':'')+v+'</span><span class="t">'+esc(o)+m+'</span></button></li>';});
    h+='<li class="somabar">Soma marcada: <b>'+(st.sel||0)+'</b></li>';
  }else if(q.type==="vf"){
    var cur=st.sel||q.opts.map(function(){return '-';}).join('');
    q.opts.forEach(function(o,k){var c="opt vfrow",ch=cur.charAt(k),corr=q.ans.charAt(k);
      if(st.done)c+=(ch===corr?" right":" wrong");
      var bt=function(x,t){var cl="vfb"+(ch===x?" on":"")+(st.done&&corr===x?" gab":"");return '<button class="'+cl+'" data-a="vf" data-k="'+k+'" data-x="'+x+'"'+(st.done?' disabled':'')+'>'+t+'</button>';};
      h+='<li><div class="'+c+'"><span class="l">'+(k+1)+'</span><span class="t">'+esc(o)+'</span><span class="vfbtns">'+bt('V','V')+bt('F','F')+'</span></div></li>';});
  }else{
  q.opts.forEach(function(o,k){
      var c="opt";
      if(st.done){if(k===q.ans)c+=" right";else if(k===st.sel)c+=" wrong";}
      else if(k===st.sel)c+=" sel";
      var m="";
      if(st.done){if(k===q.ans)m='<span class="mark">Resposta correta</span>';else if(k===st.sel)m='<span class="mark">Sua resposta</span>';}
      h+='<li><button class="'+c+'" data-a="opt" data-k="'+k+'"'+(st.done?' disabled':'')+' aria-pressed="'+(k===st.sel)+'"><span class="l">'+L[k]+'</span><span class="t">'+esc(o)+m+'</span></button></li>';});
  }
  h+='</ul>';
  if(!st.done){
    h+='<div class="actions"><button class="btn" data-a="confirm"'+(st.sel===null?' disabled':'')+'>Confirmar resposta</button></div>';
  }else{
    var ok=st.sel===q.ans;
    h+='<div class="verdict '+(ok?'ok':'no')+'" role="status"><h3>'+(ok?'Você acertou':'Você errou')+'</h3>'+
    corrHtml(q)+
    (q.ex?'<p><span class="lab">Explicação</span>'+esc(q.ex)+'</p>':'')+explainBox(q);
    if(st.err[q.id])h+='<p class="note">Esta questão fica no seu caderno de erros'+(ok?', mesmo acertando agora':'')+'. Ela só sai quando você tocar em “Tirar dos erros”.</p><div><button class="btn ghost" data-a="drop" data-id="'+q.id+'">Tirar dos erros</button></div>';
    if(q.pdf!==undefined)h+='<p class="note">O gabarito do PDF indica a letra '+L[q.pdf].toUpperCase()+' para esta questão, mas '+esc(cad.pdfNote)+'. A resposta acima é a historicamente correta.</p>';
    h+='</div><div class="actions"><button class="btn" data-a="next">'+(lastInFlow()?'Ver resultado':'Próxima')+'</button></div>';
  }
  h+='</section>';
  app.innerHTML=h;
}
function result(){

  '<p class="small">'+(all.d?Math.round(all.r/all.d*100):0)+'% de acertos em '+all.d+' questões respondidas</p>'+orn()+
  '<ul class="res" style="text-align:left">';
  AUL.forEach(function(a){var s=stats(a.n);h+='<li><span class="rom">'+a.rom+'</span><span>'+esc(a.title)+'</span><span class="idx-meta"><b>'+s.r+'</b>/'+s.d+' de '+s.t+'</span></li>';});
  h+='</ul><div class="home-actions" style="justify-content:center">'+(w.length?'<button class="btn" data-a="gorev">Refazer erros ('+w.length+')</button>':'')+'<button class="btn ghost" data-a="home">Índice</button></div></section>';

}

}

function render(){
  if(st.view==="hub")hub();
  else if(st.view==="home")home();
  else if(st.view==="inter")inter(st.i);
  else if(st.view==="q")question();
  else if(st.view==="result")result();
  else if(st.view==="revres")revResult();
  else if(st.view==="fin")finView();
}
function goto(i,withInter){
  st.mode="main";st.i=i;st.sel=null;st.done=false;
  var q=QS[i];
  var newAula=(i===0||QS[i-1].aula!==q.aula);
  st.view=(withInter&&newAula)?"inter":"q";
  render();window.scrollTo(0,0);
}
function startRev(at){
  var w=errList();if(!w.length){st.view="home";st.tab="rev";render();return;}
  st.mode="rev";st.queue=w;st.qp=at||0;st.revRight=0;st.revTotal=w.length;
  st.i=w[st.qp];st.sel=null;st.done=false;st.view="q";render();window.scrollTo(0,0);
}
function goHome(tab){st.mode="main";st.view="home";if(tab)st.tab=tab;render();window.scrollTo(0,0);}
function goHub(){var g=cad&&cad.grupo?cad.grupo:null;cad=null;curGrp=g;st=blank();document.documentElement.setAttribute('data-cad',g?grpOf(g).theme:'hub');render();window.scrollTo(0,0);}
app.addEventListener('click',function(e){
  var b=e.target.closest('[data-a]');if(!b)return;
  var a=b.getAttribute('data-a');
  if(a==="home"){goHome();}
  else if(a==="hub"){goHub();}
  else if(a==="allhub"){curGrp=null;document.documentElement.setAttribute('data-cad','hub');render();window.scrollTo(0,0);}
  else if(a==="grp"){curGrp=b.getAttribute('data-id');document.documentElement.setAttribute('data-cad',grpOf(curGrp).theme);render();window.scrollTo(0,0);}
  else if(a==="open"){loadCad(b.getAttribute('data-id'));render();window.scrollTo(0,0);}
  else if(a==="gorev"){goHome("rev");}
  else if(a==="tab"){st.tab=b.getAttribute('data-t');render();}
  else if(a==="theme"){var r=document.documentElement,cur=r.getAttribute('data-theme');
    var dark=cur?cur==="dark":window.matchMedia('(prefers-color-scheme: dark)').matches;
    r.setAttribute('data-theme',dark?'light':'dark');}
  else if(a==="go"){goto(parseInt(b.getAttribute('data-i'),10),true);}
  else if(a==="aula"){goto(firstIdx(parseInt(b.getAttribute('data-n'),10)),true);}
  else if(a==="start"){st.view="q";render();window.scrollTo(0,0);}
  else if(a==="bit"&&!st.done){var v=parseInt(b.getAttribute('data-v'),10);st.sel=(st.sel||0)^v;if(st.sel===0)st.sel=null;render();}
  else if(a==="vf"&&!st.done){var q0=QS[st.i],kk=parseInt(b.getAttribute('data-k'),10),cu=(st.sel||q0.opts.map(function(){return '-';}).join('')).split('');cu[kk]=b.getAttribute('data-x');st.sel=cu.join('');render();}
  else if(a==="explain"){var qq=QS[st.i];doExplain(qq,b);}
  else if(a==="export"){exportAll();}
  else if(a==="import"){importAll();}
  else if(a==="gsave"){var inp=b.closest('details').querySelector('.gin'),v=(inp.value||"").trim();if(v){try{localStorage.setItem(GKEY,v);}catch(e){}render();}}
  else if(a==="gdel"){try{localStorage.removeItem(GKEY);}catch(e){}render();}
  else if(a==="zoom"){b.classList.toggle("zoomed");}
  else if(a==="opt"&&!st.done){st.sel=parseInt(b.getAttribute('data-k'),10);render();}
  else if(a==="confirm"&&canConfirm(QS[st.i])&&!st.done){var q=QS[st.i];st.ans[q.id]=st.sel;st.done=true;noteResult(q,st.sel===q.ans);
    if(st.mode==="rev"){if(st.sel===q.ans)st.revRight++;}else{st.pos=Math.min(st.i+1,QS.length-1);}
    save();render();
    var v=app.querySelector('.verdict');if(v)v.scrollIntoView({block:"nearest",behavior:"smooth"});}
  else if(a==="next"){
    if(st.mode==="rev"){if(st.qp+1>=st.queue.length){st.view="revres";render();}
      else{st.qp++;st.i=st.queue[st.qp];st.sel=null;st.done=false;st.view="q";render();window.scrollTo(0,0);}}
    else{var n=st.i+1<QS.length?st.i+1:-1;if(n===-1){st.view="result";render();}else goto(n,true);}}
  else if(a==="finish"){
    if(st.mode==="main"){st.pos=Math.min(st.done?st.i+1:st.i,QS.length-1);}
    st.fin=new Date().toISOString();save();st.view="fin";render();}
  else if(a==="result"){st.view="result";render();}
  else if(a==="drop"){delete st.err[b.getAttribute('data-id')];save();render();}
  else if(a==="ftog"){var n=parseInt(b.getAttribute('data-n'),10);st.filt[n]=!picked(n);save();render();}
  else if(a==="fall"){st.filt={};save();render();}
  else if(a==="fnone"){AUL.forEach(function(u){st.filt[u.n]=false;});save();render();}
  else if(a==="rev"){startRev(0);}
  else if(a==="revone"){startRev(parseInt(b.getAttribute('data-j'),10));}
});

  if(st.view!=="q"||e.metaKey||e.ctrlKey||e.altKey)return;

  if(q.type!=="soma"&&q.type!=="vf"&&!st.done&&k>-1&&k<q.opts.length&&e.key.length===1){st.sel=k;render();}
